        window.toggleSmartView = function() {
            const metricsView = document.getElementById('smartMetricsView');
            const detailView = document.getElementById('smartDetailView');
            const toggleText = document.getElementById('toggleText');

            if (metricsView.classList.contains('hidden-content')) {
                metricsView.classList.remove('hidden-content');
                detailView.classList.add('hidden-content');
                toggleText.textContent = 'Detalle SMART';
            } else {
                metricsView.classList.add('hidden-content');

                toggleText.textContent = 'Ver Métricas';

        window.setActivePhase = function(id) {
            activePhase = id ? data.find(p => p.id === id) : null;
            render();

        function renderWheel() {
            const svg = document.getElementById('wheel-svg');
            let svgContent = '';

            data.forEach((phase, index) => {
                const startAngle = index * 90;
                const endAngle = (index + 1) * 90;
                const x1 = 50 + 48 * Math.cos((Math.PI * startAngle) / 180);
                const y1 = 50 + 48 * Math.sin((Math.PI * startAngle) / 180);
                const x2 = 50 + 48 * Math.cos((Math.PI * endAngle) / 180);
                const y2 = 50 + 48 * Math.sin((Math.PI * endAngle) / 180);
                
                const isActive = activePhase?.id === phase.id;
                const isBottomHalf = index === 1 || index === 2;
                const textRotation = isBottomHalf ? -90 : 90;

                const fillColor = isActive ? phase.color : '#ffffff';
                const strokeColor = isActive ? 'white' : '#e2e8f0';
                const strokeWidth = isActive ? "2" : "0.5";
                const textFill = isActive ? 'fill-white' : 'fill-slate-400';

                svgContent += `
                    <g class="transition-all duration-500 cursor-pointer" onclick="setActivePhase('${phase.id}')">
                        <path
                            d="M 50 50 L ${x1} ${y1} A 48 48 0 0 1 ${x2} ${y2} Z"
                            fill="${fillColor}"
                            stroke="${strokeColor}"
                            stroke-width="${strokeWidth}"
                            class="transition-all duration-300 hover:fill-slate-50"
                            style="fill: ${isActive ? phase.color : ''}"
                        ></path>
                        <text
                            x="0"
                            y="0"
                            transform="translate(50, 50) rotate(${(index * 90) + 45}) translate(33, 0) rotate(${textRotation})"
                            text-anchor="middle"
                            dominant-baseline="middle"
                            class="text-[4.5px] font-bold uppercase tracking-widest pointer-events-none transition-colors duration-300 ${textFill}"
                        >
                            ${phase.title}
                        </text>
                    </g>
                `;
            });
            
            svg.innerHTML = svgContent;
        }
